\begin{frame}{项目定位与目标}
\begin{itemize}
  \item Pharsight：基于 FinSight 的 A/H 医药投研智能体，面向高校研究与演示
  \item 两种模式：医药行业研究、上市医药公司研究；中文输出
  \item 仅作投研辅助，不构成投资或医疗建议
\end{itemize}
\begin{block}{V1 目标}
保留 FinSight 三段式主干，实现受控、可审计的医药投研流程：
\texttt{研究任务书 → 采集 → 分析 → 报告 → 发布校验}
\end{block}
\begin{itemize}
  \item 上游基线 \texttt{8fc5bce}；标准运行时 Python 3.11
\end{itemize}
\end{frame}

\begin{frame}{FinSight 实测：五类问题}
\begin{enumerate}
  \item \alert{协议}：不兼容模型原生调用格式，任务静默终止
  \item \alert{状态}：缓存与复用导致检索缺失或重复
  \item \alert{证据}：来源为自由文本，无定位与主张映射
  \item \alert{运行时}：嵌套事件循环导致进程无法退出
  \item \alert{质量}：任务完成不等于产物完整，缺少必要声明
\end{enumerate}
\begin{exampleblock}{可复用部分}
编排与检查点、共享 Memory、确定性计算、模板化写作。
\end{exampleblock}
\end{frame}

\begin{frame}{对 Pharsight 的设计启示}
\begin{itemize}
  \item \textbf{精确引用}：只允许真实来源编号，未知引用阻断发布
  \item \textbf{完整证据}：原文整本解析、分页分块，保留父来源与页码定位
  \item \textbf{失败关闭}：协议、截止日、章节完整性与声明的确定性校验
  \item \textbf{保留主干}：不替换编排，检索后端可替换
\end{itemize}
\end{frame}

\begin{frame}{能力边界}
\begin{block}{已经证明}
长 PDF 证据可定位、引用可审计，报告生成可技术闭环。
\end{block}
\begin{alertblock}{尚未证明}
单一年报不足以支撑完整投研结论；OCR 与词法检索存在局限；报告不是发布级研报。
\end{alertblock}
\begin{exampleblock}{判断}
项目已从“流程可实现”进入“证据可定位、引用可审计”阶段；下一门槛是多来源覆盖与领域复核。
\end{exampleblock}
\end{frame}
